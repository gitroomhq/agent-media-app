Idempotency test — returns rows so you see the numbers

Your first run already PASSED. The block does RAISE EXCEPTION on failure — a double-charge would have shown a red FAIL error. "Success. No rows returned" = both assertions held + rolled back. This version just makes the proof visible as a 3-row table.

Run in the Supabase SQL editor. Still BEGIN … ROLLBACK — nothing persists.

BEGIN;

CREATE TEMP TABLE _proof (step text, result jsonb, balance int) ON COMMIT DROP;

DO $$
DECLARE
    v_user uuid; v_job uuid := gen_random_uuid(); v_before int; r1 jsonb; r2 jsonb;
BEGIN
    SELECT user_id, monthly_credits_remaining + purchased_balance
      INTO v_user, v_before
      FROM public.user_credits
      WHERE monthly_credits_remaining + purchased_balance >= 2
      ORDER BY 2 DESC LIMIT 1;

    INSERT INTO _proof VALUES ('0_before', NULL, v_before);

    r1 := public.deduct_credits(v_user, 1, v_job, 'idem test call 1');
    INSERT INTO _proof VALUES ('1_first_charge', r1,
      (SELECT monthly_credits_remaining + purchased_balance FROM public.user_credits WHERE user_id = v_user));

    r2 := public.deduct_credits(v_user, 1, v_job, 'idem test call 2 (SAME job)');
    INSERT INTO _proof VALUES ('2_replay_same_job', r2,
      (SELECT monthly_credits_remaining + purchased_balance FROM public.user_credits WHERE user_id = v_user));
END $$;

SELECT * FROM _proof ORDER BY step;

ROLLBACK;
